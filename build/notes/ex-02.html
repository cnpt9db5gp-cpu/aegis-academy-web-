Mapping the Stack Frame in gdb

Summary

Before you can overwrite a return address you must know exactly where it is relative to your
buffer. This lecture produces those numbers by measurement rather than by calculation, because
the compiler inserts padding you will not have predicted.

Key ideas

  • x86-64 System V ABI: push rbp; mov rbp,rsp; sub rsp,N on entry
  • Locals sit at lower addresses than the saved RBP and return address
  • The stack grows downward, so overflowing a buffer walks upward toward ret
  • 16-byte stack alignment is enforced, so the offset is rarely a round number
  • Measure rbp - rsp and the frame contents; do not compute and hope
  • Record the numbers in your scratchpad; every later exercise needs them

The frame, drawn

HIGH ADDRESSES (older frames)
┌──────────────────────┐
│ caller frame         │
├──────────────────────┤ ← RBP
│ saved RBP      +8    │
│ RETURN ADDR    +0    │ ← the target
├──────────────────────┤
│ arg5 arg4 arg3 arg2 │
│ arg1 arg1            │
├──────────────────────┤ ← RSP
│ local variables      │
│ alignment padding    │
└──────────────────────┘ ← your buffer starts near here
LOW ADDRESSES

Terminology

Term Meaning
Prologue push rbp; mov rbp,rsp; sub rsp,N
Epilogue leave; ret
rbp Frame pointer; points at the frame base
rsp Stack pointer; points at the current top
Return address Saved RIP, popped by ret
Offset Bytes from your buffer to the return address
Alignment padding Bytes inserted to keep 16-byte alignment
Frame chain rbp → previous rbp → ... → stack base
Breaking on address b *0x401136 when symbols are absent

Kali commands

# --- the core gdb workflow, in batch mode ---
gdb -q --batch \
  -ex 'break main' -ex 'run' \
  -ex 'info registers rsp rbp rip' \
  -ex 'disassemble main' \
  -ex 'x/8gx $rbp' \
  -ex 'bt' \
  ./target-weak 'AAAA'

# --- finding a useful address without symbols ---
objdump -d ./target-weak | grep -E "<main>|<vuln>|<win>"
nm ./target-weak | grep -E " T (main|vuln|win)"

# --- the frame chain ---
gdb -q --batch -ex 'break main' -ex 'run' -ex 'x/40gx $rbp' ./target-weak

# --- follow rbp to the caller's frame ---
gdb -q --batch -ex 'break main' -ex 'run' \
  -ex 'x/4gx $rbp' -ex 'x/4gx *(long*)$rbp' ./target-weak

# --- breakpoints at instructions, not functions ---
gdb -q --batch -ex 'break *0x4011a6' -ex 'run' -ex 'info registers rip' ./target-weak

Field lab

Objective: measure the exact offset from a buffer to its return address, by two independent
methods.

mkdir -p ~/aegis/labs/ex-02 && cd ~/aegis/labs/ex-02

cp ../ex-01/target.c . 2>/dev/null || cat > target.c <<'EOF'
#include <stdio.h>
#include <string.h>
void level3(const char *in) { char b16[16]; strcpy(b16, in); printf("L3 %s\n", b16); }
void level2(const char *in) { char b24[24]; strcpy(b24, in); level3(in); }
void level1(const char *in) { char b32[32]; strcpy(b32, in); level2(in); }
int main(int argc, char **argv) {
    if (argc < 2) { fprintf(stderr, "usage: %s <string>\n", argv[0]); return 1; }
    level1(argv[1]);
    return 0;
}
EOF
gcc -Wall -g -O0 -fno-stack-protector -no-pie -z execstack target.c -o target
checksec --file=./target | grep -E "No canary|NX disabled|No PIE"

# --- 1. the prologue, read directly from the disassembly ---
echo "=== the prologue of each function ==="
objdump -d -M intel ./target | grep -A6 -E "<(level1|level2|level3)>:" | head -30 | tee 01-prologue.txt
echo
echo "Read: push rbp / mov rbp,rsp / sub rsp,N.  N is the frame size for that function."

# --- 2. measure rbp - rsp in each frame ---
echo
echo "=== rbp - rsp, per frame ==="
for fn in main level1 level2 level3; do
  gdb -q --batch -ex "break $fn" -ex "run $(python3 -c 'print("A"*120)')" \
      -ex "printf \"  $fn: rbp=0x%lx rsp=0x%lx delta=%ld bytes\\n\", \$rbp, \$rsp, \$rbp-\$rsp" \
      -ex quit ./target 2>/dev/null | grep "$fn:"
done | tee 02-deltas.txt

# --- 3. THE FRAME CONTENTS: where the return address actually is ---
echo
echo "=== the frame at level1, in quads ==="
gdb -q --batch -ex 'break level1' -ex 'run AAAAAAAAAAAAAAAA' \
    -ex 'printf "rbp = 0x%lx\n", $rbp' \
    -ex 'x/8gx $rbp-16' \
    -ex 'x/2gx $rbp' \
    -ex quit ./target 2>/dev/null | tee 03-frame.txt
cat 03-frame.txt
echo
echo "  At \$rbp+0 is the SAVED RBP. At \$rbp+8 is the RETURN ADDRESS."
echo "  Identify the return address: it is a 0x0040xxxx value, not ASCII."

# --- 4. brute-force the offset (the reliable method) ---
echo
echo "=== measuring the offset empirically ==="
python3 - <<'PY' | tee 04-offset.txt
import subprocess
marker = b"ZZZZZZZZ"          # 0x5a5a5a5a5a5a5a5a
for n in range(1, 160):
    try:
        r = subprocess.run(["./target", b"A" * n + marker], capture_output=True, timeout=5)
        out = (r.stderr + r.stdout).decode("utf8", "replace")
    except subprocess.TimeoutExpired:
        continue
    if marker.decode() in out or r.returncode != 0:
        print(f"  offset to return address : {n + 8} bytes")
        print(f"  crash occurred at n      : {n}")
        print(f"  exit code                : {r.returncode}")
        print(f"  last output              : {out.strip().splitlines()[-1][:80] if out.strip() else '(none)'}")
        break
else:
    print("  no crash observed in the swept range - widen it")
PY

# --- 5. the three functions, three offsets ---
echo
echo "=== per-function offsets (each frame is different) ==="
cat > levels.c <<'EOF'
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
void l3(char *in){ char b[16]; strcpy(b,in); printf("L3 %s\n", b); }
void l2(char *in){ char b[24]; strcpy(b,in); l3(in); }
void l1(char *in){ char b[32]; strcpy(b,in); l2(in); }
int main(int argc,char**argv){
  if(argc<2) return 1;
  void *fns[3] = {l1,l2,l3};
  int which = atoi(argv[2]) if0;   /* placeholder */
  return 0; }
EOF
cat > offsets.py <<'PY'
import subprocess, sys
OFFSETS = {"l1": None, "l2": None, "l3": None}
marker = b"BBBBBBBB"
def build(which):
    src = f'''
#include <stdio.h>
#include <string.h>
void l3(char *in){{ char b[16]; strcpy(b,in); }}
void l2(char *in){{ char b[24]; strcpy(b,in); if({which}==3) l3(in); }}
void l1(char *in){{ char b[32]; strcpy(b,in); if({which}>=2) l2(in); }}
int main(int argc,char**argv){{ if(argc<2) return 1;
  if({which}==1) l1(argv[1]); else if({which}==2) l2(argv[1]); else l3(argv[1]);
  return 0; }}
'''
    open(f"/tmp/t{which}.c","w").write(src)
    subprocess.run(["gcc","-g","-O0","-fno-stack-protector","-no-pie","-z","execstack",
                    f"/tmp/t{which}.c","-o",f"/tmp/t{which}"],check=True)
for which in (1,2,3):
    build(which)
    for n in range(1,200):
        try:
            r = subprocess.run([f"/tmp/t{which}", b"A"*n + marker], capture_output=True, timeout=5)
        except subprocess.TimeoutExpired:
            continue
        if r.returncode != 0 or marker.decode() in (r.stdout+r.stderr).decode("utf8","replace"):
            OFFSETS[f"l{which}"] = n + 8
            break
    print(f"  l{which}: offset to return address = {OFFSETS[f'l{which}']} bytes")
PY
python3 offsets.py 2>/dev/null | tee 05-level-offsets.txt
rm -f levels.c offsets.py /tmp/t?.c /tmp/t?

|---|---|---|---|

Notes:
