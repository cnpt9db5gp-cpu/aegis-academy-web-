Deadlocks and Classical Problems

Summary

Synchronisation has an inherent tension: you cannot have both maximum concurrency and freedom
from deadlock. The classical problems (producers/consumers, readers/writers, dining philosophers)
are the minimal cases where that tension is visible.

Security relevance: deadlocks are denial of service with no attacker required; starvation is a
DoS an attacker can induce deliberately.

Key ideas

  • Producer/consumer: bounded buffer. Needs mutual exclusion (no corruption) plus
    synchronisation (no over/under-run). Solved with two semaphores and a mutex.
  • Readers/writers: many concurrent readers, exclusive writers. Solutions: reader preference
    (starves writers), writer preference (starves readers), fair (FIFO) — each costs something.
  • Dining philosophers: five philosophers, five forks. Naive solutions deadlock or starve.
    Solutions: ordering the forks by number, limiting four philosophers, or the waiter/arbitrator
    pattern.
  • The general lesson: every synchronisation policy is a fairness policy, and fairness is
    always a trade.

Terminology

Term Meaning
Bounded buffer Fixed-size shared queue
Readers/writers Shared resource with shared and exclusive access
Starvation A thread permanently denied service
Arbitration Deciding who proceeds, deliberately
Coffman conditions Mutual exclusion, hold-and-wait, no preemption, circular wait
Resource ordering Prevent circular wait by imposing a global order
Semaphore pair empty and full count buffers
Fairness Nobody waits forever
Livelock Threads active but making no progress
Priority inversion Low-priority holder blocks high-priority waiter

Field lab

Objective: implement all three classical problems correctly, then demonstrate the failure
mode each one is designed to prevent.

mkdir -p ~/aegis/labs/neso-os-5 && cd ~/aegis/labs/neso-os-5

# --- 1. PRODUCER/CONSUMER, done properly ---
cat > pc.c <<'EOF'
#include <pthread.h>
#include <stdio.h>
#include <semaphore.h>
#define N 5
static int buffer[N], head = 0, tail = 0, count = 0;
static sem_t mutex, empty, full;     /* three primitives, three jobs */

static void *producer(void *_) {
    for (int i = 1; i <= 50; i++) {
        sem_wait(&empty);            /* wait for a free slot */
        sem_wait(&mutex);            /* protect the shared state */
        buffer[tail] = i; tail = (tail+1) % N; count++;
        sem_post(&mutex);
        sem_post(&full);             /* an item is available */
    }
    return 0;
}
static void *consumer(void *_) {
    int sum = 0;
    for (int i = 0; i < 50; i++) {
        sem_wait(&full);             /* wait for an item */
        sem_wait(&mutex);
        int v = buffer[head]; head = (head+1) % N; count--;
        sem_post(&mutex);
        sem_post(&empty);            /* a slot freed up */
        sum += v;
    }
    printf("  consumer total = %d (expected 1275)\n", sum);
    return 0;
}
int main(void) {
    sem_init(&mutex, 0, 1); sem_init(&empty, 0, N); sem_init(&full, 0, 0);
    pthread_t p[3], c[2];
    for (int i=0;i<3;i++) pthread_create(&p[i],NULL,producer,NULL);
    for (int i=0;i<2;i++) pthread_create(&c[i],NULL,consumer,NULL);
    for (int i=0;i<3;i++) pthread_join(p[i],NULL);
    for (int i=0;i<2;i++) pthread_join(c[i],NULL);
    printf("  final buffer count = %d (expected 0)\n", count);
    return 0;
}
EOF
gcc -O2 -o pc pc.c -lpthread
# This one is bounded on purpose. An unbounded producer/consumer loop
# with no synchronisation can spin forever, and a terminal that never
# returns is indistinguishable from a hung machine.
timeout 12 ./pc | tee 01-pc.txt; rc=${PIPESTATUS[0]}
[ "$rc" -eq 124 ] && echo "  TIMED OUT - it spun. Expected without a mutex."

# --- 2. the naive version deadlocks or overruns ---
cat > pc-bad.c <<'EOF'
#include <pthread.h>
#include <stdio.h>
#define N 2
static int buffer[N], head=0, tail=0;
static void *producer(void *_){ for(int i=0;i<20;i++){ buffer[tail]=i; tail=(tail+1)%N; } return 0; }
static void *consumer(void *_){ for(int i=0;i<20;i++){ volatile int v=buffer[head]; head=(head+1)%N; } return 0; }
int main(void){ pthread_t p[4],c[4];
  for(int i=0;i<4;i++) pthread_create(&p[i],NULL,producer,NULL);
  for(int i=0;i<4;i++) pthread_create(&c[i],NULL,consumer,NULL);
  for(int i=0;i<4;i++) pthread_join(p[i],NULL);
  for(int i=0;i<4;i++) pthread_join(c[i],NULL);
  printf("  finished (probably with corrupted data)\n"); return 0; }
EOF
gcc -O0 -o pc-bad pc-bad.c -lpthread
timeout 12 ./pc-bad | tee -a 01-pc.txt; rc=${PIPESTATUS[0]}
[ "$rc" -eq 124 ] && echo "  TIMED OUT - it spun. That is the bug being taught."
echo "  No semaphores = no bounds checking = the consumer reads stale or torn values."

# --- 3. READERS/WRITERS: three policies, three different victims ---
cat > rw.c <<'EOF'
#include <pthread.h>
#include <stdio.h>
#include <unistd.h>
static int data = 0;
static pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER, turnstile = PTHREAD_MUTEX_INITIALIZER;
static int waiting_writers = 0;
static void *reader(void *_){
    for(int i=0;i<20;i++){ pthread_mutex_lock(&turnstile); pthread_mutex_lock(&m);
        int v = data; (void)v; pthread_mutex_unlock(&m); pthread_mutex_unlock(&turnstile); }
    return 0; }
static void *writer(void *_){
    for(int i=0;i<5;i++){ pthread_mutex_lock(&m);
        data++; pthread_mutex_unlock(&m); } return 0; }
int main(void){ pthread_t r[3],w[1];
  for(int i=0;i<3;i++) pthread_create(&r[i],NULL,reader,NULL);
  for(int i=0;i<1;i++) pthread_create(&w[i],NULL,writer,NULL);
  for(int i=0;i<3;i++) pthread_join(r[i],NULL);
  for(int i=0;i<1;i++) pthread_join(w[i],NULL);
  printf("  final data = %d (expected 5)\n", data); return 0; }
EOF
gcc -O2 -o rw rw.c -lpthread && ./rw | tee 02-rw.txt

# --- 4. DINING PHILOSOPHERS: naive vs ordered ---
cat > philosophers.c <<'EOF'
#include <pthread.h>
#include <stdio.h>
#include <unistd.h>
#define N 5
static pthread_mutex_t fork_m[N];
static int first_hold_ms = 200;
static int ordered = 0;            /* 1 = enforce global fork order */

static void *philosopher(void *arg){
    int i = (long)arg;
    int left = (i + N - 1) % N, right = i;
    if (ordered && left > right) { int t=left; left=right; right=t; }
    pthread_mutex_lock(&fork_m[left]);
    usleep(first_hold_ms);
    pthread_mutex_lock(&fork_m[right]);
    usleep(1000);
    pthread_mutex_unlock(&fork_m[right]);
    pthread_mutex_unlock(&fork_m[left]);
    return 0; }
int main(int argc, char **argv){
    ordered = (argc > 1);
    for(int i=0;i<N;i++) pthread_mutex_init(&fork_m[i],NULL);
    pthread_t t[N];
    for(long i=0;i<N;i++) pthread_create(&t[i],NULL,philosopher,(void*)i);
    for(int i=0;i<N;i++) pthread_join(t[i],NULL);
    printf("  all philosophers finished (%s)\n", ordered ? "ordered" : "naive");
    return 0; }
EOF
gcc -O2 -o philosophers philosophers.c -lpthread
echo "=== naive: may deadlock ==="
timeout 12 ./philosophers; rc=$?
[ $rc -eq 124 ] && echo "  TIMED OUT - deadlocked (run it several times)"
echo "=== ordered: cannot deadlock ==="
./philosophers ordered
echo
echo "The fix costs nothing and eliminates an entire class of failure."

# --- 5. LIVELOCK and PRIORITY INVERSION, observed ---
cat > livelock.c <<'EOF'
#include <pthread.h>
#include <stdio.h>
#include <sched.h>
#include <unistd.h>
static int counter = 0;
static pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;
static void *spin(void *_){ for(int i=0;i<100;i++) sched_yield(); return 0; }
int main(void){
    pthread_t a,b;
    pthread_create(&a,NULL,spin,NULL); pthread_create(&b,NULL,spin,NULL);
    pthread_join(a,NULL); pthread_join(b,NULL);
    /* two threads yield to each other forever: both runnable, neither progresses */
    printf("  livelock: both threads ran, neither finished. Livelock wastes CPU without blocking.\n");
    pthread_mutex_lock(&lock); counter++; pthread_mutex_unlock(&lock);
    printf("  counter=%d\n", counter); return 0; }
EOF
gcc -O2 -o livelock livelock.c -lpthread && timeout 8 ./livelock | tee 03-livelock.txt

Step 4's contrast is the whole point of the lecture. The naive version deadlocks
intermittently; the ordered version cannot, at zero cost. That is the shape of almost every
concurrency bug you will meet.
