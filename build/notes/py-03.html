Strings, Files and Reading Data

Summary

Security work is text processing: parsing nmap output, reading logs, extracting hashes from
/etc/shadow, building wordlists. Python makes this short and safe, and the with statement
makes it reliable.

The most important idea: stream vs whole-file. A 4 GB memory dump will crash a script that
reads it all at once.

File modes

Mode Meaning
r read (default, text)
w write, truncates the file first
a append
x write, fails if the file exists
rb / wb binary — required for pcaps, images, binaries

w destroys the file immediately. A diagnostic script opening the wrong path in w mode can
destroy the very evidence you are investigating.

with is not optional

with open("/etc/passwd") as f:          # closes even if an exception occurs
    for line in f:
        print(line.strip())

f = open("/etc/passwd")                 # works, but leaks the handle on exception
...
f.close()

In security tooling, the with failure mode is "you exhausted your file descriptors after 1000
iterations". Use it every time.

Stream, don't slurp

# BAD - a 4 GB memory image becomes a 4 GB Python string
data = open("memdump.raw").read()

# GOOD - constant memory regardless of file size
with open("memdump.raw", "rb") as f:
    while chunk := f.read(1024 * 1024):   # walrus: assign and test in one step
        process(chunk)

Line-oriented streaming, the common case:

with open("/var/log/auth.log", errors="replace") as f:
    for line in f:                      # never fully loaded
        if "Failed password" in line:
            print(line.strip())

Reading binary data

with open("sample.bin", "rb") as f:
    header = f.read(16)
    print(header.hex())     # hex dump - your first malware triage step
    print(header[:4])       # magic bytes

SIGNATURES = {
    b"\x7fELF":   "ELF executable",
    b"MZ":        "Windows PE executable",
    b"PK\x03\x04": "Zip archive",
    b"\x89PNG":   "PNG image",
    b"%PDF":      "PDF document",
    b"\xd0\xcf\x11\xe0": "OLE2 (legacy Office)",
}
kind = next((v for k, v in SIGNATURES.items() if header.startswith(k)), "unknown")
print(kind)

That is file(1) in six lines. Knowing magic bytes is how you identify a payload that has
been renamed — one of the oldest tricks in the book.

Parsing for security work

Encoding: the eternal footgun

# ALWAYS pass errors= - one bad byte otherwise kills the script
text = data.decode("utf8", errors="replace")
with open("suspect.log", errors="replace") as f:
    lines = f.readlines()

In Python 3 the default encoding comes from your locale. Never assume; pass it explicitly.

Terminology

Term Meaning
Text vs binary mode 'r' decodes to str; 'rb' gives raw bytes
Encoding The byte↔character mapping (UTF-8, latin-1)
with Context manager guaranteeing cleanup
Stream Read incrementally without loading the whole file
Magic bytes Leading bytes identifying a file type
Walrus := Assign-and-test in one expression
Decode vs encode bytes→str is decode; str→bytes is encode

# --- file(1) in Python ---
cat > identify.py <<'PY'
import sys, pathlib
SIGNATURES = [
    (b"\x7fELF", "ELF executable"), (b"MZ", "Windows PE executable"),
    (b"PK\x03\x04", "Zip / Office / JAR"), (b"\x89PNG", "PNG image"),
    (b"\xff\xd8\xff", "JPEG image"), (b"GIF8", "GIF image"),
    (b"%PDF", "PDF document"), (b"\x1f\x8b", "gzip stream"),
    (b"BZh", "bzip2"), (b"\xfd7zXZ", "xz"), (b"7z\xbc\xaf\x27\x1c", "7-zip"),
    (b"Rar!", "RAR archive"), (b"\xd0\xcf\x11\xe0", "OLE2 (legacy Office)"),
    (b"\xca\xfe\xba\xbe", "Java class / Mach-O fat"),
    (b"SQLite format 3", "SQLite database"),
]
def identify(path):
    with open(path, "rb") as f:
        head = f.read(4096)
    for magic, name in SIGNATURES:
        if head.startswith(magic):
            return name
    return "data"
if __name__ == "__main__":
    for arg in sys.argv[1:]:
        p = pathlib.Path(arg)
        if p.is_file():
            print(f"{str(p):<40} {p.stat().st_size:>10}  {identify(str(p))}")
PY
python3 identify.py /usr/bin/nmap /bin/bash /etc/passwd 2>/dev/null | tee 01-identify.txt
file /usr/bin/nmap /bin/bash /etc/passwd | tee -a 01-identify.txt

# --- a log parser that streams ---
cat > logparse.py <<'PY'
import re, sys, collections

PATTERNS = {
    "failed_ssh":   re.compile(r"Failed password for (?:invalid user )?(?P<user>\S+) from (?P<ip>[\d.]+)"),
    "accepted_ssh": re.compile(r"Accepted password for (?P<user>\S+) from (?P<ip>[\d.]+)"),
    "sudo":         re.compile(r"sudo:\s+(\S+)\s*:.*COMMAND=(?P<cmd>.+)"),
}
def parse(path):
    counts, attackers, users = collections.Counter(), collections.Counter(), collections.Counter()
    with open(path, errors="replace") as f:
        for line in f:
            if m := PATTERNS["failed_ssh"].search(line):
                counts["failed_ssh"] += 1
                attackers[m.group("ip")] += 1
                users[m.group("user")] += 1
            elif m := PATTERNS["accepted_ssh"].search(line):
                counts["accepted_ssh"] += 1
                users[m.group("user")] += 1
            elif m := PATTERNS["sudo"].search(line):
                counts["sudo"] += 1
    return counts, attackers, users
if __name__ == "__main__":
    c, a, u = parse(sys.argv[1] if len(sys.argv) > 1 else "auth-sample.log")
    print(f"events: {dict(c)}\n")
    print("top source IPs:")
    for ip, n in a.most_common(10): print(f"  {ip:<20} {n}")
    print("\ntop targeted usernames:")
    for name, n in u.most_common(10): print(f"  {name:<20} {n}")
PY
python3 logparse.py auth-sample.log | tee 02-logparse.txt

Step 6 is the engineering lesson. Streaming 300 MB costs constant small memory; reading it
whole costs 300 MB. On a 4 GB memory image that difference decides whether your tool works.
