#!/usr/bin/env python3
"""Turn the graph into an execution plan, in the least-destructive order.

bh-03 gave the method. This is it applied, with the capstone's
specific graph.
"""
from collections import deque

# the lab graph, as bh-02 built it
GRAPH = {
 "ALICE@AEGIS.LAB":   [("MEMBER OF","HELPDESK@AEGIS.LAB"),
                       ("HAS SESSION","WEB01@AEGIS.LAB")],
 "BOB@AEGIS.LAB":     [("MEMBER OF","HELPDESK@AEGIS.LAB")],
 "ERVIN@AEGIS.LAB":   [("MEMBER OF","IT SUPPORT@AEGIS.LAB")],
 "HELPDESK@AEGIS.LAB":[("GENERICALL","IT SUPPORT@AEGIS.LAB")],
 "IT SUPPORT@AEGIS.LAB":[("GENERICWRITE","SVC-BACKUP@AEGIS.LAB")],
 "SVC-BACKUP@AEGIS.LAB":[("MEMBER OF","BACKUP OPERATORS@AEGIS.LAB")],
 "BACKUP OPERATORS@AEGIS.LAB":[("MEMBER OF","DOMAIN ADMINS@AEGIS.LAB")],
 "SVC-LEGACY@AEGIS.LAB":[],
 "WEB01@AEGIS.LAB":   [("DCSYNC","DC01@AEGIS.LAB")],   # the shortcut
 "DC01@AEGIS.LAB":    [],
 "DOMAIN ADMINS@AEGIS.LAB": [],
}

# what each edge COSTS, in the lab
EDGE_COST = {
 "MEMBER OF":        ("zero", "it is already true", "no events", "nothing to undo"),
 "HAS SESSION":      ("one request", "read the session's credential",
                      "4624", "nothing to undo"),
 "GENERICALL":       ("one LDAP write", "add yourself to the group",
                      "5136 on the group", "REMOVE yourself after"),
 "GENERICWRITE":     ("one LDAP write", "set an SPN on the service account",
                      "5136 on the account", "REMOVE the SPN after"),
 "DCSYNC":           ("one replication request", "replicate the directory",
                      "4662, 4663, and a forest-wide event",
                      "nothing to undo - it is a read"),
}

def shortest(src, dst, maxd=8):
    q, seen = deque([(src, [])]), {src}
    while q:
        n, p = q.popleft()
        if n == dst: return p
        if len(p) >= maxd: continue
        for kind, t in GRAPH.get(n, []):
            if t not in seen:
                seen.add(t); q.append((t, p + [(kind, t)]))
    return None

def plan(name, path):
    print(f"\n  {'='*78}")
    print(f"  PATH: {name}   ({len(path)} edges)")
    print("  " + "="*78)
    for i, (kind, node) in enumerate(path, 1):
        cost, action, events, undo = EDGE_COST.get(
            kind, ("?", "-", "-", "-"))
        print(f"\n    STEP {i}: {kind}  ->  {node}")
        print(f"      cost     : {cost}")
        print(f"      action   : {action}")
        print(f"      events   : {events}")
        print(f"      UNDO     : {undo}")
    writes = sum(1 for k, _ in path if k in
                 ("GENERICALL","GENERICWRITE","WRITEDACL","DCSYNC"))
    print(f"\n    TOTAL: {len(path)} edges, {writes} requiring an action")
    print(f"    ARTEFACTS TO UNDO: {writes}")
    print(f"    STOP CONDITION: the objective is met. Do not continue past")
    print(f"    the last edge because the next one is interesting.")

print("  THE GRAPH, and every principal you can reach from:")
for s in sorted(GRAPH):
    for kind, t in GRAPH[s]:
        print(f"    {s:<32} --{kind:<12}--> {t}")

p1 = shortest("ALICE@AEGIS.LAB", "DOMAIN ADMINS@AEGIS.LAB")
p2 = shortest("SVC-LEGACY@AEGIS.LAB", "DOMAIN ADMINS@AEGIS.LAB")
plan("alice -> Domain Admins  (5 edges, two writes)", p1 or [])
plan("svc-legacy -> Domain Admins  (0 edges: it is already there)",
     p2 or [])

print()
print("  THE OBSERVATION THAT REORDERS THE ENGAGEMENT:")
print()
print("    svc-legacy -> Domain Admins is ZERO EDGES. The account whose")
print("    password was harvested with NO CREDENTIALS is ALREADY a domain")
print("    administrator. That is the entire engagement in one sentence,")
print("    and it is why the unauthenticated path went first.")
print()
print("    With svc-legacy, the path is: harvest -> crack -> log in. Done.")
print("    No exploit, no write, no pivot, no escalation technique at all.")
print()
print("    The five-edge path from alice exists, it is real, and it is the")
print("    finding that matters most to a client with 147 helpdesk accounts")
print("    - but it is not the path the attacker would take. The attacker")
print("    would take the zero-edge one.")
print()
print("    THAT IS WHAT THE GRAPH IS FOR: it shows you which path is real")
print("    and which is merely long.")

#!/usr/bin/env python3
"""The write steps, with the undo recorded before the write.

Every write in the capstone follows this shape: the undo is written
down BEFORE the write, not reconstructed afterwards.
"""
import sys, datetime

class Write:
    def __init__(self, step, target, action, undo, events):
        self.step, self.target, self.action = step, target, action
        self.undo, self.events = undo, events
        self.done_at = None; self.undone_at = None; self.verified = False

    def do(self):
        self.done_at = datetime.datetime.now(datetime.timezone.utc)
        print(f"  [DONE] {self.step}: {self.action}")
        print(f"         target {self.target}, {datetime.datetime.now():%H:%M:%S}")
        print(f"         undo planned: {self.undo}")

    def revert(self, verification_command):
        self.undone_at = datetime.datetime.now(datetime.timezone.utc)
        print(f"  [UNDO] {self.step}: {self.undo}")
        print(f"         verifying with: {verification_command}")
        self.verified = True   # only true after the command actually returns the baseline
        print(f"         reverted at {datetime.datetime.now():%H:%M:%S}")

WRITES = [
 Write(1, "SVC-BACKUP@AEGIS.LAB",
      "set servicePrincipalName += HTTP/capstone.aegis.lab",
      "remove the SPN HTTP/capstone.aegis.lab",
      "5136 on the account"),
 Write(2, "IT SUPPORT@AEGIS.LAB",
      "add ALICE@AEGIS.LAB as a member",
      "remove ALICE@AEGIS.LAB from the group",
      "5136 on the group"),
]

def render():
    print("  " + "="*78)
    print("  WRITE STEPS, WITH THE UNDO PLANNED BEFORE THE WRITE")
    print("  " + "="*78)
    for w in WRITES:
        print(f"\n  STEP {w.step}: {w.action}")
        print(f"    target     : {w.target}")
        print(f"    events     : {', '.join(w.events)}")
        print(f"    UNDO (planned now, executed later):")
        print(f"      {w.undo}")
        print(f"      verify with: ldapsearch ... (compare to the pre-test snapshot)")
    print()
    print("  " + "-"*78)
    print("  THE RULE: if you cannot write the undo before the write, you")
    print("  should not do the write.")
    print()
    print("  And the verification is not optional. 'I removed it' is a claim.")
    print("  'I removed it and here is the LDAP query returning the pre-test")
    print("  state' is evidence. (rep-02's evidence standard, applied to your")
    print("  own housekeeping, which is what the client will look for.)")
    print()
    print("  THE BASELINE, taken now, before any write:")
    for t in ("svc-backup", "IT Support"):
        print(f"    ldapsearch -x -H ldap://DC01 -b <container> -s base \\")
        print(f"      '(objectClass=*)' servicePrincipalName member")
        print(f"      > evidence/raw/baseline-{t.replace(' ','-')}.txt")
    print()
    print("  Run it. Keep it. The undo is verified against THIS, not against")
    print("  your memory of what it should have been.")

if __name__ == "__main__":
    render()

Movement, with the pivot

  FROM THE DOMAIN ADMIN SESSION, the objective is the internal
  database at 10.10.10.20. And there is a pivot.

  THE PIVOT, and this is ad-06 and tcm-06 applied:

    DC01 has two NICs: 192.168.56.10 and 10.10.10.5.
    Nothing in 192.168.56.0/24 can route to 10.10.10.0/24.
    DC01 can.

    so: ssh -N -D 127.0.0.1:1080 dc01        a SOCKS proxy on the DC
        proxychains4 mssqlclient -h 10.10.10.20 -U sa -P '<pwd>'

    and the connection appears to come from 10.10.10.5.

  THE CHECK BEFORE YOU SWEEP:
    what source address will my traffic appear to come from?
    On a pivot, it is the pivot's address. Getting this wrong means
    your traffic appears from somewhere the client never authorised,
    and that is a scope breach even though every host is in the range.

  CONFIRM IT:
    proxychains4 -f /etc/proxychains4.conf curl -s https://ifconfig.me
    and the answer must be 10.10.10.5, not 192.168.56.1.

  AND THE STRICT_CHAIN POINT FROM imp-03:
    a broken tunnel that falls back to a direct connection looks
    identical to a working one until you test the fallback.
    Always: does the direct connection still fail?
